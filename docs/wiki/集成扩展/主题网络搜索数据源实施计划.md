> **给执行者：** 按任务顺序逐条执行。每个任务都以「先写测试 → 确认失败 → 最小实现 → 确认通过」推进。
> 设计依据：[主题网络搜索数据源设计.md](./主题网络搜索数据源设计.md)（若实现中发现设计与代码冲突，先改设计文档再改代码）。
> **本计划不含 git 提交步骤**——遵循项目约定"未经明确要求不提交"，如需提交请人工确认。

**目标：** 在现有数据源同步框架上新增 `topic_search` 连接器，实现"按自定义主题定时搜索公网 → 抓取网页正文 → 自动入库并解析"，仅后端。

**架构：** 复用现有 cron 调度、asynq 任务、同步日志、游标持久化与 `ingestItem → CreateKnowledgeFromFile` 入库链路；新增的连接器只负责"搜索 + 抓正文 + 两级变更检测"，依赖三个包内最小接口（搜索 / 默认 Provider 解析 / 网页抓取），全部可注入假实现。唯一侵入性改动是把 `ProcessSync` 中的租户上下文注入提前到 fetch 之前。

**技术栈：** Go 1.26 · robfig/cron（复用）· asynq（复用）· `internal/infrastructure/web_fetch`（复用）· `internal/application/service.WebSearchService`（复用）· 标准库 `httptest`/假实现做单测

// Package topicsearch 实现「主题网络搜索」数据源连接器。
//
// 它按用户配置的主题在公网搜索，抓取搜索结果的网页正文并入库：每个主题是一个
// 可勾选的资源，每个网页是一条知识条目，external_id 为规范化后的 URL，因此重复
// 同步时正文未变的页面会被跳过、正文变化的页面会被更新。
//
// 变更检测分两级（与 RSS 连接器同构）：搜索结果可见字段（标题 + 摘要 + 发布时间）
// 未变时完全不抓取页面；变了才抓取，再比较正文哈希。
//
// 出网请求由注入的抓取器（web_fetch，SSRF 安全）完成；黑名单过滤由注入的
// WebSearchService 负责，连接器不重复实现，避免两套规则漂移。
package topicsearch

import (
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"net/url"
	"strconv"
	"strings"
	"time"
	"unicode/utf8"

	"github.com/Tencent/WeKnora/internal/datasource"
	"github.com/Tencent/WeKnora/internal/types"
)

// 默认值与上限。数值越界一律夹紧（不报错），避免用户改配置即同步失败。
const (
	defaultMaxResultsPerTopic = 10
	defaultMaxItemsPerRun     = 5
	defaultMinContentLength   = 200

	maxTopics             = 20
	maxResultsPerTopicCap = 50
	maxItemsPerRunCap     = 50
	maxRecencyDays        = 365
	maxBlacklistRules     = 200
	maxFileNameBytes      = 200
)

// trackingParams 是从规范化 URL 中剔除的常见追踪参数（业务参数一律保留）。
var trackingParams = map[string]struct{}{
	"gclid": {}, "fbclid": {}, "msclkid": {}, "yclid": {}, "spm": {}, "scm": {}, "_ga": {},
}

// Config 是 topic_search 的连接器配置，全部为非密钥配置（存于 DataSourceConfig.Settings）。
type Config struct {
	Topics             []string
	ProviderID         string
	MaxResultsPerTopic int
	MaxItemsPerRun     int
	MinContentLength   int
	ResultRecencyDays  int
	Blacklist          []string
}

// parseConfig 解析并夹紧连接器配置；topics 为空视为配置无效。
func parseConfig(config *types.DataSourceConfig) (*Config, error) {
	if config == nil {
		return nil, fmt.Errorf("%w: config is nil", datasource.ErrInvalidConfig)
	}
	settings := config.Settings
	cfg := &Config{
		Topics:             settingStringList(settings, "topics"),
		ProviderID:         settingString(settings, "provider_id"),
		MaxResultsPerTopic: settingInt(settings, "max_results_per_topic", defaultMaxResultsPerTopic),
		MaxItemsPerRun:     settingInt(settings, "max_items_per_run", defaultMaxItemsPerRun),
		MinContentLength:   settingInt(settings, "min_content_length", defaultMinContentLength),
		ResultRecencyDays:  settingInt(settings, "result_recency_days", 0),
		Blacklist:          settingStringList(settings, "blacklist"),
	}
	cfg.normalize()
	if len(cfg.Topics) == 0 {
		return nil, fmt.Errorf("%w: topics is required", datasource.ErrInvalidConfig)
	}
	return cfg, nil
}

// normalize 夹紧数值、限制列表长度、去重。
func (c *Config) normalize() {
	c.Topics = capList(c.Topics, maxTopics)
	c.Blacklist = capList(c.Blacklist, maxBlacklistRules)
	c.MaxResultsPerTopic = clamp(c.MaxResultsPerTopic, 1, maxResultsPerTopicCap)
	c.MaxItemsPerRun = clamp(c.MaxItemsPerRun, 1, maxItemsPerRunCap)
	if c.MinContentLength < 0 {
		c.MinContentLength = 0
	}
	c.ResultRecencyDays = clamp(c.ResultRecencyDays, 0, maxRecencyDays)
}

// searchConfig 构造传给 WebSearchService 的搜索参数。
// IncludeDate 仅在需要按时效过滤时打开，避免无谓的 Provider 参数变化。
func (c *Config) searchConfig() *types.WebSearchConfig {
	return &types.WebSearchConfig{
		MaxResults:  c.MaxResultsPerTopic,
		IncludeDate: c.ResultRecencyDays > 0,
		Blacklist:   c.Blacklist,
	}
}

// topicsFor 返回本轮要同步的主题：resourceIDs 为空表示全部，否则按配置顺序取交集。
func (c *Config) topicsFor(resourceIDs []string) []string {
	if len(resourceIDs) == 0 {
		return c.Topics
	}
	selected := make(map[string]struct{}, len(resourceIDs))
	for _, id := range resourceIDs {
		if id = strings.TrimSpace(id); id != "" {
			selected[id] = struct{}{}
		}
	}
	out := make([]string, 0, len(c.Topics))
	for _, topic := range c.Topics {
		if _, ok := selected[topic]; ok {
			out = append(out, topic)
		}
	}
	return out
}

// withinRecency 按时效窗口过滤搜索结果；窗口关闭或结果无发布时间时一律保留。
func (c *Config) withinRecency(publishedAt *time.Time, now time.Time) bool {
	if c.ResultRecencyDays <= 0 || publishedAt == nil || publishedAt.IsZero() {
		return true
	}
	return publishedAt.After(now.AddDate(0, 0, -c.ResultRecencyDays))
}

// contentFingerprint 对正文做短哈希，用于增量变更检测。
func contentFingerprint(content string) string {
	sum := sha256.Sum256([]byte(content))
	return "h:" + hex.EncodeToString(sum[:])[:16]
}

// signalFingerprint 对搜索结果可见字段做短哈希：未变则无需抓取页面。
func signalFingerprint(result *types.WebSearchResult) string {
	if result == nil {
		return ""
	}
	var b strings.Builder
	b.WriteString(strings.TrimSpace(result.Title))
	b.WriteByte('\n')
	b.WriteString(strings.TrimSpace(result.Snippet))
	b.WriteByte('\n')
	if result.PublishedAt != nil && !result.PublishedAt.IsZero() {
		b.WriteString(result.PublishedAt.UTC().Format(time.RFC3339))
	}
	sum := sha256.Sum256([]byte(b.String()))
	return "s:" + hex.EncodeToString(sum[:])[:16]
}

// normalizeURL 生成用于去重的稳定 URL：小写 scheme/host、去 fragment、去追踪参数、
// query 按 key 排序。非 http(s) 或缺少 host 时返回错误。
func normalizeURL(raw string) (string, error) {
	raw = strings.TrimSpace(raw)
	if raw == "" {
		return "", fmt.Errorf("empty url")
	}
	u, err := url.Parse(raw)
	if err != nil {
		return "", fmt.Errorf("parse url: %w", err)
	}
	scheme := strings.ToLower(u.Scheme)
	if scheme != "http" && scheme != "https" {
		return "", fmt.Errorf("unsupported scheme %q", u.Scheme)
	}
	if u.Host == "" {
		return "", fmt.Errorf("missing host")
	}
	u.Scheme = scheme
	u.Host = strings.ToLower(u.Host)
	u.Fragment = ""
	u.RawQuery = normalizeQuery(u.Query())
	return u.String(), nil
}

// normalizeQuery 去掉追踪参数；url.Values.Encode 按 key 排序，结果天然稳定。
func normalizeQuery(values url.Values) string {
	filtered := make(url.Values, len(values))
	for key, v := range values {
		lower := strings.ToLower(key)
		if strings.HasPrefix(lower, "utm_") {
			continue
		}
		if _, ok := trackingParams[lower]; ok {
			continue
		}
		filtered[key] = v
	}
	return filtered.Encode()
}

// sanitizeFileName 去掉文件名非法字符并按 UTF-8 边界截断（与 RSS 连接器一致）。
func sanitizeFileName(name string) string {
	name = strings.TrimSpace(name)
	if name == "" {
		return "untitled"
	}
	replacer := strings.NewReplacer(
		"/", "_", "\\", "_", ":", "_", "*", "_",
		"?", "_", "\"", "_", "<", "_", ">", "_", "|", "_",
		"\n", " ", "\r", " ", "\t", " ",
	)
	result := strings.TrimSpace(replacer.Replace(name))
	if result == "" {
		return "untitled"
	}
	if len(result) > maxFileNameBytes {
		result = result[:maxFileNameBytes]
		for len(result) > 0 {
			r, size := utf8.DecodeLastRuneInString(result)
			if r != utf8.RuneError || size != 1 {
				break
			}
			result = result[:len(result)-1]
		}
	}
	return result
}

// settingString 读取字符串型配置，非字符串或缺失时返回空串。
func settingString(settings map[string]interface{}, key string) string {
	if settings == nil {
		return ""
	}
	s, ok := settings[key].(string)
	if !ok {
		return ""
	}
	return strings.TrimSpace(s)
}

// settingInt 宽松解析数值：接受 JSON 数字与数字字符串，非法值回落到默认值。
func settingInt(settings map[string]interface{}, key string, def int) int {
	if settings == nil {
		return def
	}
	switch v := settings[key].(type) {
	case float64:
		return int(v)
	case int:
		return v
	case int64:
		return int(v)
	case string:
		parsed, err := strconv.Atoi(strings.TrimSpace(v))
		if err != nil {
			return def
		}
		return parsed
	default:
		return def
	}
}

// settingStringList 接受「多行/逗号分隔字符串」或「字符串数组」两种形态。
func settingStringList(settings map[string]interface{}, key string) []string {
	if settings == nil {
		return nil
	}
	switch v := settings[key].(type) {
	case string:
		return splitList(v)
	case []string:
		return dedupeStrings(v)
	case []interface{}:
		items := make([]string, 0, len(v))
		for _, raw := range v {
			if s, ok := raw.(string); ok {
				items = append(items, s)
			}
		}
		return dedupeStrings(items)
	default:
		return nil
	}
}

// splitList 按换行/逗号切分并去重（保持原顺序）。
func splitList(raw string) []string {
	fields := strings.FieldsFunc(raw, func(r rune) bool {
		return r == '\n' || r == '\r' || r == ','
	})
	return dedupeStrings(fields)
}

// dedupeStrings 去空白、去重，保持首次出现的顺序。
func dedupeStrings(values []string) []string {
	seen := make(map[string]struct{}, len(values))
	out := make([]string, 0, len(values))
	for _, v := range values {
		v = strings.TrimSpace(v)
		if v == "" {
			continue
		}
		if _, ok := seen[v]; ok {
			continue
		}
		seen[v] = struct{}{}
		out = append(out, v)
	}
	return out
}

func clamp(value, minValue, maxValue int) int {
	if value < minValue {
		return minValue
	}
	if value > maxValue {
		return maxValue
	}
	return value
}

func capList(values []string, max int) []string {
	if len(values) <= max {
		return values
	}
	return values[:max]
}

```go
// walk 是 FetchAll / FetchIncremental 的共享实现。
//
// incremental 为 true 时启用两级变更检测：搜索结果可见字段（signal）未变则完全不
// 抓取；变了才抓取并比较正文哈希。新游标记录本轮已处理的页面（含判定为稳定或过短
// 的页面），未处理的页面（含因配额中断）不写入，留给下一轮。
func (c *Connector) walk(
	ctx context.Context,
	config *types.DataSourceConfig,
	resourceIDs []string,
	prev *topicCursor,
	incremental bool,
) ([]types.FetchedItem, *topicCursor, error) {
	cfg, err := parseConfig(config)
	if err != nil {
		return nil, nil, err
	}
	providerID, err := c.resolveProviderID(ctx, cfg)
	if err != nil {
		return nil, nil, err
	}
	if c.fetcher == nil {
		return nil, nil, fmt.Errorf("%w: 缺少网页抓取器", datasource.ErrInvalidConfig)
	}

	topics := cfg.topicsFor(resourceIDs)
	newCursor := &topicCursor{
		LastSyncTime: time.Now().UTC(),
		Pages:        map[string]cursorEntry{},
	}

	var (
		out          []types.FetchedItem
		warnings     []string
		seenURLs     = map[string]struct{}{}
		produced     int
		failedTopics int
	)

	for _, topic := range topics {
		if produced >= cfg.MaxItemsPerRun {
			break
		}
		results, searchErr := c.search.Search(ctx, providerID, cfg.searchConfig(), topic)
		if searchErr != nil {
			failedTopics++
			logger.Warnf(ctx, "[TopicSearch] search topic %q failed: %v", topic, searchErr)
			warnings = append(warnings, fmt.Sprintf("topic %q: %v", topic, searchErr))
			continue
		}

		for _, result := range results {
			if produced >= cfg.MaxItemsPerRun {
				break
			}
			if result == nil || strings.TrimSpace(result.URL) == "" {
				continue
			}
			normURL, normErr := normalizeURL(result.URL)
			if normErr != nil {
				logger.Warnf(ctx, "[TopicSearch] skip invalid url %q: %v", result.URL, normErr)
				continue
			}
			if _, dup := seenURLs[normURL]; dup {
				continue
			}
			seenURLs[normURL] = struct{}{}
			if !cfg.withinRecency(result.PublishedAt, time.Now().UTC()) {
				continue
			}

			signal := signalFingerprint(result)
			var prevEntry *cursorEntry
			if incremental && prev != nil {
				if entry, ok := prev.Pages[normURL]; ok {
					prevEntry = &entry
				}
			}
			// 第一级：搜索结果可见字段未变 → 不抓取，直接把状态带到新游标
			if prevEntry != nil && prevEntry.Signal != "" && prevEntry.Signal == signal {
				newCursor.Pages[normURL] = *prevEntry
				continue
			}

			content, fetchErr := c.fetcher.Fetch(ctx, result.URL)
			if fetchErr != nil {
				// 不写游标：下一轮重试该页面
				logger.Warnf(ctx, "[TopicSearch] fetch %s failed: %v", result.URL, fetchErr)
				warnings = append(warnings, fmt.Sprintf("page %s: %v", result.URL, fetchErr))
				continue
			}
			content = strings.TrimSpace(content)
			hash := contentFingerprint(content)
			entry := cursorEntry{
				Signal: signal,
				Hash:   hash,
				Topic:  topic,
				At:     time.Now().UTC().Format(time.RFC3339),
			}

			// 第二级：正文未变 → 跳过（游标刷新 signal，避免下轮重复抓取）
			if prevEntry != nil && prevEntry.Hash != "" && prevEntry.Hash == hash {
				newCursor.Pages[normURL] = entry
				continue
			}
			// 正文过短：不入库，但记录游标，避免每轮重复抓取
			if utf8.RuneCountInString(content) < cfg.MinContentLength {
				newCursor.Pages[normURL] = entry
				logger.Infof(ctx, "[TopicSearch] skip short page %s (%d chars)",
					result.URL, utf8.RuneCountInString(content))
				continue
			}

			newCursor.Pages[normURL] = entry
			out = append(out, buildItem(result, normURL, topic, providerID, content))
			produced++
		}
	}

	if len(topics) > 0 && failedTopics == len(topics) {
		return nil, newCursor, fmt.Errorf("all topics failed: %s", strings.Join(warnings, "; "))
	}
	if len(warnings) > 0 {
		return out, newCursor, &datasource.PartialFetchError{Details: warnings}
	}
	return out, newCursor, nil
}

// buildItem 把抓取到的正文组装成知识条目。
func buildItem(
	result *types.WebSearchResult, normURL, topic, providerID, content string,
) types.FetchedItem {
	title := strings.TrimSpace(result.Title)
	if title == "" {
		title = fallbackTitle(result.URL)
	}
	updatedAt := time.Now().UTC()
	metadata := map[string]string{
		"channel":         types.ChannelTopicSearch,
		"topic":           topic,
		"source_url":      result.URL,
		"search_provider": providerID,
	}
	if result.PublishedAt != nil && !result.PublishedAt.IsZero() {
		updatedAt = *result.PublishedAt
		metadata["published_at"] = result.PublishedAt.UTC().Format(time.RFC3339)
	}
	return types.FetchedItem{
		ExternalID:       normURL,
		Title:            title,
		Content:          []byte(content),
		ContentType:      "text/markdown",
		FileName:         sanitizeFileName(title) + ".md",
		URL:              result.URL,
		UpdatedAt:        updatedAt,
		SourceResourceID: topic,
		Metadata:         metadata,
	}
}

// fallbackTitle 在搜索结果无标题时用 host + path 生成可读标题。
func fallbackTitle(rawURL string) string {
	u, err := url.Parse(rawURL)
	if err != nil || u.Host == "" {
		return "untitled"
	}
	path := strings.Trim(u.Path, "/")
	if path == "" {
		return u.Host
	}
	return u.Host + "/" + path
}

	ctx = context.WithValue(ctx, types.TenantIDContextKey, ds.TenantID)

	tenant, err := s.tenantRepo.GetTenantByID(ctx, ds.TenantID)
	if err != nil {
		logger.Errorf(ctx, "failed to get tenant info: %v", err)
		syncLog.Status = types.SyncLogStatusFailed
		syncLog.FinishedAt = timePtr(time.Now().UTC())
		syncLog.ErrorMessage = fmt.Sprintf("Failed to get tenant info: %v", err)
		_ = s.syncLogRepo.Update(ctx, syncLog)
		if !wasPaused {
			ds.Status = types.DataSourceStatusError
		}
		ds.ErrorMessage = syncLog.ErrorMessage
		_ = s.dsRepo.Update(ctx, ds)
		return err
	}
	ctx = context.WithValue(ctx, types.TenantInfoContextKey, tenant)
```

注意：`tenant` 变量随后被使用的位置不变（同一作用域），移动后仍可编译。
